# Eviction Filings, 2020 - 2025, all
filings_20_25 <- dbGetQuery(con, "
  SELECT DISTINCT(i.indexnumberid), fileddate, court
  FROM oca_index i
  WHERE i.fileddate >= MAKE_DATE(2020,1,1) 
      AND i.fileddate <= MAKE_DATE(2025,12,31)
      AND i.classification IN ('Holdover', 'Non-Payment')
      AND (i.propertytype = 'Residential' OR i.propertytype IS NULL);
")

# Eviction Filings, 2020 - 2025, with location
filings_20_25_sf <- st_read(con, query = "
  SELECT DISTINCT(i.indexnumberid), fileddate, court,
    ST_Transform(a.geom, 2263) AS geom
  FROM oca_index i
  JOIN oca_addresses a ON i.indexnumberid = a.indexnumberid
  WHERE i.fileddate >= MAKE_DATE(2020,1,1) 
      AND i.fileddate <= MAKE_DATE(2025,12,31)
      AND i.classification IN ('Holdover', 'Non-Payment')
      AND (i.propertytype = 'Residential' OR i.propertytype IS NULL)
      AND a.geom IS NOT NULL;
")

# Executed Evictions, 2020 - 2025, all
executions_20_25 <- dbGetQuery(con, "
  SELECT DISTINCT(i.indexnumberid), w.executiondate, i.court
  FROM oca_index i
  JOIN oca_warrants w ON i.indexnumberid = w.indexnumberid
  WHERE w.executiondate >= MAKE_DATE(2020,1,1) 
    AND w.executiondate <= MAKE_DATE(2025,12,31)
    AND i.classification IN ('Holdover', 'Non-Payment')
    AND (i.propertytype = 'Residential' OR i.propertytype IS NULL)
    AND w.executiondate IS NOT NULL;
")

# Executed Evictions, 2020 - 2025, with location
executions_20_25_sf <- st_read(con, query = "
  WITH executed_evictions AS (
    SELECT DISTINCT(i.indexnumberid), w.executiondate, i.court
    FROM oca_index i
    JOIN oca_warrants w ON i.indexnumberid = w.indexnumberid
    WHERE w.executiondate >= MAKE_DATE(2020,1,1) 
      AND w.executiondate <= MAKE_DATE(2025,12,31)
      AND i.classification IN ('Holdover', 'Non-Payment')
      AND (i.propertytype = 'Residential' OR i.propertytype IS NULL)
      AND w.executiondate IS NOT NULL
  )
  SELECT e.*, ST_TRANSFORM(a.geom, 2263)
  FROM executed_evictions e
  JOIN oca_addresses a on e.indexnumberid = a.indexnumberid
  WHERE a.geom IS NOT NULL
")

                                   month(executiondate) < 4 ~ TRUE,
                                   month(executiondate) == 4 & day(executiondate) <= 15 ~ TRUE,
                                   .default = FALSE
                                   ))

### Assembly Districts ###
nysad_compiled <- nysad |> 
  select(District, Name, Party) |> 
  mutate(Region = case_when(District %in% nysad_upstate$District ~ "Upstate",
                            District %in% nysad_downstate$District ~ "Downstate",
                            .default = NA))

executions_20_25_nysad <- executions_20_25_sf |> 
  st_join(nysad_compiled |> select(District))



nysad_compiled <- nysad_compiled |> 
  left_join(
    bind_rows(
      nysad_downstate_5year,
      nysad_upstate_5year
    ) |> 
      rename("filings_20_25" = "Eviction Filings") |> 
      select(!Name),
    by = join_by(District)
  ) |> 
  left_join(
    bind_rows(
      nysad_downstate_1year,
      nysad_upstate_1year
    ) |> 
      rename("filings_25" = "Eviction Filings") |> 
      select(!Name),
    by = join_by(District)
  ) |> 
  left_join(executions_20_25_nysad |> 
              st_drop_geometry() |> 
              group_by(District) |> 
              summarize(executions_20_25 = n())) |> 
  left_join(executions_20_25_nysad |> 
              filter(year(executiondate) == 2025) |> 
              st_drop_geometry() |> 
              group_by(District) |> 
              summarize(executions_25 = n())) |> 
  left_join(filings_20_25_sf |> 
              filter(winter_season == TRUE) |> 
              st_join(nysad_compiled |> select(District)) |>
              st_drop_geometry() |> 
              group_by(District) |> 
              summarize(winter_filings_20_25 = n())) |> 
  left_join(filings_20_25_sf |> 
              filter(winter_season == TRUE & year(fileddate) == 2025) |> 
              st_join(nysad_compiled |> select(District)) |>
              st_drop_geometry() |> 
              group_by(District) |> 
              summarize(winter_filings_25 = n())) |> 
  left_join(executions_20_25_nysad |> 
              filter(winter_season == TRUE) |> 
              st_drop_geometry() |> 
              group_by(District) |> 
              summarize(winter_executions_20_25 = n())) |> 
  left_join(executions_20_25_nysad |> 
              filter(winter_season == TRUE & year(executiondate) == 2025) |> 
              st_drop_geometry() |> 
              group_by(District) |> 
              summarize(winter_executions_25 = n())) |> 
  st_drop_geometry()

# Rearrange
nysad_compiled <- nysad_compiled |> 
  relocate(contains("20_25"),
           .after = Region)

### NYC Council ###
nyc_council <- nyc_council |> 
  select(CounDist, name) |> 
  arrange(CounDist)

filings_20_25_nycc <- filings_20_25_sf |> 
  st_join(nyc_council |> select(CounDist)) |> 
  st_drop_geometry()

executions_20_25_nycc <- executions_20_25_sf |> 
  st_join(nyc_council |> select(CounDist)) |> 
  st_drop_geometry()

nyc_council_compiled <- nyc_council |> 
  # Filings 20-25
  left_join(filings_20_25_nycc |> 
              filter(!is.na(CounDist)) |> 
              group_by(CounDist) |> 
              summarize(filings_20_25 = n())) |> 
  # Executions 20-25
  left_join(executions_20_25_nycc |> 
              filter(!is.na(CounDist)) |> 
              group_by(CounDist) |> 
              summarize(executions_20_25 = n())) |> 
  # Winter filings 20-25
  left_join(filings_20_25_nycc |> 
              filter(!is.na(CounDist) & winter_season == TRUE) |> 
              group_by(CounDist) |> 
              summarize(winter_filings_20_25 = n())) |> 
  # Winter Executions 20-25
  left_join(executions_20_25_nycc |> 
              filter(!is.na(CounDist) & winter_season == TRUE) |> 
              group_by(CounDist) |> 
              summarize(winter_executions_20_25 = n())) |> 
  # Filings 25
  left_join(filings_20_25_nycc |> 
              filter(!is.na(CounDist) & year(fileddate) == 2025) |> 
              group_by(CounDist) |> 
              summarize(filings_25 = n())) |>
  # Executions 25
  left_join(executions_20_25_nycc |> 
              filter(!is.na(CounDist) & year(executiondate) == 2025) |> 
              group_by(CounDist) |> 
              summarize(executions_25 = n())) |> 
  # Winter filings 25
  left_join(filings_20_25_nycc |> 
              filter(!is.na(CounDist) & winter_season == TRUE & year(fileddate) == 2025) |> 
              group_by(CounDist) |> 
              summarize(winter_filings_25 = n())) |>
  #Winter Executions 25
  left_join(executions_20_25_nycc |> 
              filter(!is.na(CounDist) & winter_season == TRUE & year(executiondate) == 2025) |> 
              group_by(CounDist) |> 
              summarize(winter_executions_25 = n())) |> 
  st_drop_geometry()

write_xlsx(list("NYS Assembly" = nysad_compiled,

                "NYC Council" = nyc_council_compiled),
          path = "NYS_Evictions_2020_2025.xlsx")
